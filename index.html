from ursina import * from ursina.prefabs.first_person_controller import FirstPersonController from perlin_noise import PerlinNoise import random app = Ursina() # 🌤️ НАСТРОЙКА ОКНА И ФОНА window.color = color.rgb(135, 206, 235) # Стабильный голубой фон вместо баганого неба scene.fog_density = 0 # Без тумана, чтобы не отбеливать экран # Генератор шума для создания бесконечного холмистого рельефа noise = PerlinNoise(octaves=2, seed=random.randint(1, 1000)) # 🌍 ГЕНЕРАЦИЯ БОЛЬШОГО МИРА WORLD_SIZE = 35 for x in range(-WORLD_SIZE, WORLD_SIZE, 2): for z in range(-WORLD_SIZE, WORLD_SIZE, 2): # Рассчитываем плавную высоту гор и холмов y = int(noise([x * 0.05, z * 0.05]) * 12) # Красим мир в зависимости от высоты if y < -2: block_color = color.rgb(222, 184, 135) # Песчаные пляжи elif y > 3: block_color = color.rgb(112, 128, 144) # Скалы на вершинах else: block_color = color.rgb(46, 139, 87) # Зеленая трава # Создаем 3D куб земли Entity( model='cube', color=block_color, position=(x, y, z), scale=(2, 4, 2), # Вытянутая форма, чтобы под землей не было дыр collider='box', shader=None ) # 🧍 ИГРОК player = FirstPersonController(position=(0, 12, 0), origin_y=-0.5) # 🚗 ДЕТАЛИЗИРОВАННАЯ МАШИНА (Теперь это объемный спорткар, а не кнопки!) car = Entity( model='cube', color=color.red, scale=(2.5, 1, 4.5), position=(5, 8, 5), collider='box', shader=None ) # Элементы дизайна машины (родительские объекты привязаны к корпусу) cabin = Entity(parent=car, model='cube', color=color.rgb(30, 30, 30), scale=(0.8, 0.6, 0.5), position=(0, 0.7, -0.2)) # Кабина wheel1 = Entity(parent=car, model='cube', color=color.black, scale=(0.3, 0.6, 0.6), position=(0.55, -0.4, 1.3)) # Переднее правое wheel2 = Entity(parent=car, model='cube', color=color.black, scale=(0.3, 0.6, 0.6), position=(-0.55, -0.4, 1.3)) # Переднее левое wheel3 = Entity(parent=car, model='cube', color=color.black, scale=(0.3, 0.6, 0.6), position=(0.55, -0.4, -1.3)) # Заднее правое wheel4 = Entity(parent=car, model='cube', color=color.black, scale=(0.3, 0.6, 0.6), position=(-0.55, -0.4, -1.3))# Заднее левое # Переменные для физики и состояния машины is_driving = False car_speed = 0 max_speed = 25 gravity = 0 # Глобальный флаг для принудительного меню / освобождения мыши menu_mode = False # 🎮 ИГРОВОЙ ЦИКЛ ОБНОВЛЕНИЯ (Каждый кадр) def update(): global is_driving, car_speed, gravity, menu_mode # 1. ОБРАБОТКА РЕЖИМА СВОБОДНОЙ МЫШКИ (Кнопка Ctrl) if menu_mode: mouse.locked = False player.cursor.visible = True return # Блокируем ходьбу и езду, пока мышь активна else: player.cursor.visible = False # 2. ФИЗИКА ГРАВИТАЦИИ ДЛЯ МАШИНЫ (Чтобы она не висела в воздухе) # Пускаем луч вниз под машину для определения земли ray = raycast(car.position + Vec3(0, 0.5, 0), Vec3(0, -1, 0), ignore=(car,)) if ray.hit and ray.distance < 1.1: gravity = 0 car.y = ray.point.y + 0.5 # Ставим машину четко на поверхность холма else: gravity -= time.dt * 15 # Падение вниз car.y += gravity * time.dt # 3. УПРАВЛЕНИЕ МАШИНОЙ if is_driving: if held_keys['w']: car_speed = min(car_speed + time.dt * 15, max_speed) elif held_keys['s']: car_speed = max(car_speed - time.dt * 15, -max_speed / 2) else: car_speed *= 0.95 # Плавная остановка (инерция) if held_keys['a']: car.rotation_y -= time.dt * car_speed * 3.5 if held_keys['d']: car.rotation_y += time.dt * car_speed * 3.5 # Перемещение машины вперед по направлению ее капота car.position += car.forward * car_speed * time.dt # Камера плавно следует за машиной сверху-сзади (Вид от 3-го лица) player.position = car.position + Vec3(0, 2.5, 0) - car.forward * 8 player.camera_pivot.rotation_x = 15 player.rotation_y = car.rotation_y # ⌨️ СИСТЕМА НАЖАТИЯ КЛАВИШ def input(key): global is_driving, menu_mode # ПРОВЕРКА НАЖАТИЯ НА CTRL (Левый или Правый) if key == 'control' or key == 'left control': menu_mode = not menu_mode if not menu_mode: mouse.locked = True # Возвращаем мышь в игру # Посадка и высадка из машины на клавишу 'F' if key == 'f' and not menu_mode: if not is_driving: # Проверяем расстояние между игроком и машиной if distance(player.position, car.position) < 6: is_driving = True player.visible = False # Скрываем игрока внутри авто car.collider = None # Отключаем коллизию кузова, чтоб машина не застревала else: is_driving = False player.visible = True player.position = car.position + car.left * 2.5 + Vec3(0, 1.5, 0) # Высаживаем игрока слева car.collider = 'box' player.camera_pivot.rotation_x = 0 # Сбрасываем наклон камеры # Закрытие игры if key == 'escape': application.quit() app.run()
